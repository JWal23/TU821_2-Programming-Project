Basic code to check for an input without enter key:
```
#include <iostream>
#include <conio.h>
#include <string>

int main ()
{
    std::cout <<"press any key";
    while (true) {
    int key = _getch();
    if (key == 'h' || key == 'H') {
        std::cout << "Hello World\n";}
    else if (key == 'q' || key == 'Q') {
    break;
    } else {
        std::cout << "You pressed: " << key << "\n";
    }
    
    return 0;
    }
}
```
